package main

import "fmt"

func main() {
	var closures []func() int
	var pointers []*int
	for _, value := range []int{1, 2, 3} {
		closures = append(closures, func() int { return value })
		pointers = append(pointers, &value)
	}
	// Every closure runs after the loop; no scheduler timing is involved.
	fmt.Println("closures:", call(closures))
	fmt.Println("pointers:", []int{*pointers[0], *pointers[1], *pointers[2]})
	fmt.Println("pointers equal:", pointers[0] == pointers[1] && pointers[1] == pointers[2])

	closures = nil
	var value int
	for _, value = range []int{1, 2, 3} {
		closures = append(closures, func() int { return value })
	}
	fmt.Println("assignment closures:", call(closures))

	shared := []int{1}
	var rows [][]int
	for _, row := range [][]int{shared, shared, shared} {
		rows = append(rows, row)
	}
	// Distinct iteration variables still copy slice descriptors, not their data.
	rows[0][0] = 99
	fmt.Println("shared backing array:", []int{rows[0][0], rows[1][0], rows[2][0]})
}

func call(closures []func() int) []int {
	var values []int
	for _, closure := range closures {
		values = append(values, closure())
	}
	return values
}
